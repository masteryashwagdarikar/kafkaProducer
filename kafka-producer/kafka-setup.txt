STEP 1 START KAFKA : 
1. set JAVA_HOME=C:\Program Files\Java\jdk-27
2. "%JAVA_HOME%\bin\java" -Xmx512M -server -XX:+UseG1GC -XX:MaxGCPauseMillis=20 -XX:G1IHOP=35 -cp "libs\*" kafka.Kafka config\server.properties


STEP 2 CREATE TOPIC :
cd C:\kafka
bin\windows\kafka-topics.bat --create --topic quickstart-events --bootstrap-server localhost:9092


STEP 3 START KAFKA CONSUMER:
 C:\>cd kafka

C:\kafka>set JAVA_HOME=C:\Program Files\Java\jdk-27

C:\kafka>bin\windows\kafka-console-consumer.bat --topic quickstart-events --from-beginning --bootstrap-server localhost:9092


STEP 4 START KAFKA PRODUCER:
C:\kafka>set JAVA_HOME=C:\Program Files\Java\jdk-27

C:\kafka>bin\windows\kafka-console-producer.bat --topic quickstart-events --bootstrap-server localhost:9092
